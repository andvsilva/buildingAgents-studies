\subsection*{5.5 Example Structure}

\begin{minted}{python}
class QuantumState:
    def __init__(self, state_vector):
        self.state_vector = state_vector

    def normalize(self):
        norm = sum(abs(x)**2 for x in self.state_vector) ** 0.5
        self.state_vector = [x / norm for x in self.state_vector]
\end{minted}
